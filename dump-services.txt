/* ------------------------------------------------------------------
   Smart search
   ------------------------------------------------------------------
   One entry point for the whole search bar. The UI calls smartSearch()
   and renders whatever comes back. It never needs to know which
   Supabase function produced a result.

   Pipeline:
     query -> parse_search_query (subject + intent)
           -> search_medicines / search_drug_classes / search_interactions
           -> one normalised result object
------------------------------------------------------------------ */

// RxCT deliberately has no dosing content. Dosing errors can be fatal.
export const DOSAGE_NOTICE =
  "RxCT does not include dosing. Dosing errors can cause serious harm, " +
  "so check the BNF, the product label, or a pharmacist for doses.";

// A medicine counts as a confident match at or above this score.
// Below it, the UI should show "Did you mean" instead of jumping in.
const CONFIDENT_SCORE = 0.75;

// Which of the 5 medicine sections a search intent should open first.
// There is no dosing section, so "dosage" falls back to the overview.
export const INTENT_TO_SECTION = {
  overview: "overview",
  indications: "indications",
  adverse_effects: "adverse_effects",
  contraindications: "contraindications",
  interactions: "interactions",
  dosage: "overview",
};

export async function parseSearchQuery(q) {
  return executeRequest(
    "parseSearchQuery",
    async () => {
      const { data, error } = await supabase.rpc("parse_search_query", { q });

      if (error) throw error;
      return data?.[0] ?? null;
    },
    { query: q },
  );
}

// search_drug_classes returns one flat row per (class, medicine).
// This folds those rows into { type, title, slug, classes: [...] }.
export function groupClassRows(rows) {
  if (!rows || rows.length === 0) return null;

  const first = rows[0];
  const isGroup = first.result_type === "group";
  const byClass = new Map();

  for (const row of rows) {
    if (!byClass.has(row.drug_class)) {
      byClass.set(row.drug_class, []);
    }

    byClass.get(row.drug_class).push({
      id: row.medicine_id,
      name: row.medicine_name,
    });
  }

  return {
    type: isGroup ? "group" : "classes",
    title: isGroup ? first.group_name : null,
    slug: isGroup ? first.group_slug : null,
    classes: [...byClass].map(([drugClass, medicines]) => ({
      drugClass,
      medicines,
    })),
    medicineCount: rows.length,
  };
}

/*
  smartSearch(query) always resolves to this shape:

  {
    kind:         "medicines" | "classes" | "none",
    query:        what the user typed,
    subject:      the query with intent words stripped ("warfarin"),
    intent:       "overview" | "indications" | "adverse_effects" |
                  "contraindications" | "interactions" | "dosage",
    openSection:  which medicine section to open first,
    notice:       a message to show above results, or null,
    didYouMean:   true when the medicine matches are weak,
    medicines:    [{ id, name, drug_class, matched_term, score }],
    classResult:  output of groupClassRows(), or null,
    interactions: [{ medicine_id, medicine_name, drug_class, snippet }],
  }
*/
export async function smartSearch(rawQuery) {
  const query = (rawQuery ?? "").trim();

  const empty = {
    kind: "none",
    query,
    subject: "",
    intent: "overview",
    openSection: "overview",
    notice: null,
    didYouMean: false,
    medicines: [],
    classResult: null,
    interactions: [],
  };

  if (!query) return empty;

  const parsed = await parseSearchQuery(query);

  const subject = parsed?.subject ?? query;
  const intent = parsed?.intent ?? "overview";
  const wantsClass = parsed?.wants_class ?? false;

  const base = {
    ...empty,
    subject,
    intent,
    openSection: INTENT_TO_SECTION[intent] ?? "overview",
    notice: intent === "dosage" ? DOSAGE_NOTICE : null,
  };

  // Everything was an intent word ("dosage" on its own): nothing to search.
  if (!subject) return base;

  const [medicines, classRows, interactions] = await Promise.all([
    searchMedicines(query),
    searchDrugClasses(query),
    intent === "interactions" ? searchInteractions(query) : Promise.resolve([]),
  ]);

  const classResult = groupClassRows(classRows);
  const topScore = medicines[0]?.score ?? 0;
  const hasConfidentMedicine = topScore >= CONFIDENT_SCORE;

  let kind = "none";

  if (wantsClass && classResult) {
    kind = "classes";
  } else if (hasConfidentMedicine) {
    kind = "medicines";
  } else if (classResult) {
    kind = "classes";
  } else if (medicines.length > 0) {
    kind = "medicines";
  }

  return {
    ...base,
    kind,
    didYouMean: kind === "medicines" && !hasConfidentMedicine,
    medicines,
    classResult,
    interactions,
  };
}

  String(a.name).toLowerCase() === String(b.name).toLowerCase();

      ...history.filter((entry) => !same(entry, item)),
    ].slice(0, HISTORY_LIMIT),
  );
}

  try {
    const value = localStorage.getItem(key);
    return value ? JSON.parse(value) : fallback;
  } catch {
    return fallback;
  }
};

const write = (key, value) => {
  try {
    localStorage.setItem(key, JSON.stringify(value));

    return true;
  } catch {
    return false;
  }
};

const entityKey = (context) =>
  context ? `${context.type}:${context.name.toLowerCase()}` : "general";

export function addNote(context, text) {
  const all = read(NOTES_KEY, {});
  const key = entityKey(context);

  const collection = all[key] || {
    type: context?.type || "general",
    name: context?.name || "General",
    className: context?.className || null,
    groupName: context?.groupName || null,
    entries: [],
  };

  collection.entries.unshift({
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    text,
    section: context?.sectionLabel || null,
    createdAt: new Date().toISOString(),
  });

  all[key] = collection;

  return write(NOTES_KEY, all);
}

/* ========================================
   LANDING PAGE DATA
   Classes (with how many medicines each
   holds and which group they belong to)
   and the therapeutic groups.
======================================== */

const FEATURED_LIMIT = 5;

const normalise = (value = "") => String(value).trim().toLowerCase();

export async function getLandingData() {
  return executeRequest("getLandingData", async () => {
    const [classResult, groupResult] = await Promise.all([
      supabase
        .from("drug_classes")
        .select("id, name, slug, medicines(name)")
        .order("name"),

      supabase
        .from("drug_class_groups")
        .select(
          "id, name, sort_order, drug_class_group_members(drug_class)",
        )
        .order("sort_order"),
    ]);

    if (classResult.error) throw classResult.error;
    if (groupResult.error) throw groupResult.error;

    // class name -> the first group that contains it
    const groupOfClass = new Map();

    for (const group of groupResult.data) {
      for (const member of group.drug_class_group_members || []) {
        const key = normalise(member.drug_class);

        if (!groupOfClass.has(key)) {
          groupOfClass.set(key, group.name);
        }
      }
    }

    const classes = classResult.data
      .map((row) => {
        const names = (row.medicines || [])
          .map((medicine) => medicine.name)
          .sort((a, b) => a.localeCompare(b));

        return {
          id: row.id,
          name: row.name,
          medicineCount: names.length,
          members: names.slice(0, 3),
          groupName: groupOfClass.get(normalise(row.name)) || "",
        };
      })
      .filter((item) => item.medicineCount > 0);

    return { classes };
  });
}

/* ========================================
   WHICH CLASSES TO FEATURE
   Restrained on purpose: classes the
   student explored recently come first,
   then the classes with the most medicines
   in the catalogue. Never more than five.
======================================== */

export function pickFeaturedClasses(classes, recentActivity = []) {
  const recentNames = recentActivity
    .filter((item) => item.type === "class")
    .map((item) => normalise(item.name));

  const recentRank = (item) => {
    const index = recentNames.indexOf(normalise(item.name));

    return index === -1 ? Infinity : index;
  };

  return [...classes]
    .sort((a, b) => {
      const byRecent = recentRank(a) - recentRank(b);

      // Infinity - Infinity is NaN, so compare explicitly
      if (recentRank(a) !== recentRank(b)) return byRecent < 0 ? -1 : 1;

      if (b.medicineCount !== a.medicineCount) {
        return b.medicineCount - a.medicineCount;
      }

      return a.name.localeCompare(b.name);
    })
    .slice(0, FEATURED_LIMIT);
}

/* ========================================
   READING STATE
   Remembers, per medicine / class / group:
   the active section, scroll position per
   section and which subsections are open.

   Stored on the device so it also works
   offline. Moves to IndexedDB with the
   offline phase; only this file changes.

const STORAGE_KEY = "rxct-reading-state-v1";
const MAX_ENTRIES = 60;
const SAVE_DELAY = 400;

let cache = null;
let timer = 0;

const load = () => {
  if (cache) return cache;

  try {
    cache = JSON.parse(localStorage.getItem(STORAGE_KEY)) || {};
  } catch {
    cache = {};
  }

  return cache;
};

const persist = () => {
  const all = load();
  const keys = Object.keys(all);

  if (keys.length > MAX_ENTRIES) {
    keys
      .sort((a, b) => (all[a].updatedAt || 0) - (all[b].updatedAt || 0))
      .slice(0, keys.length - MAX_ENTRIES)
      .forEach((key) => delete all[key]);
  }

  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(all));
  } catch {
    // Storage may be unavailable or full.
  }
};

const schedulePersist = () => {
  clearTimeout(timer);
  timer = setTimeout(persist, SAVE_DELAY);
};

export function getReadingState(key) {
  return load()[key] || null;
}

export function saveReadingState(key, patch) {
  const all = load();
  const current = all[key] || {};

  all[key] = {
    ...current,
    ...patch,
    scroll: { ...current.scroll, ...patch.scroll },
    expanded: { ...current.expanded, ...patch.expanded },
    updatedAt: Date.now(),
  };

  schedulePersist();
}

if (typeof window !== "undefined") {
  window.addEventListener("pagehide", () => {
    clearTimeout(timer);
    persist();
  });
}

    requestId: generateRequestId(),
    timestamp: new Date().toISOString(),

      status: error?.status || error?.statusCode || null,
      message: error?.message || "Unknown error",

      hint: error?.hint || null,

  };

  if (import.meta.env.DEV) {

  }

  return {
    ...diagnostic,
    userMessage: getUserErrorMessage(category),
  };

import { reportError } from "./reportError";

export async function executeRequest(operation, request, context = {}) {
  const startedAt = Date.now();

  try {
    return await request();
  } catch (error) {
    const diagnostic = reportError({
      operation,
      error,
      durationMs: Date.now() - startedAt,
      query: context.query || null,
    });

    const appError = new Error(error?.message || "Request failed", {
      cause: error,
    });

    Object.assign(appError, {
      code: error?.code,
      status: error?.status || error?.statusCode,
      details: error?.details,
      hint: error?.hint,
      diagnostic,
    });

    throw appError;
  }
}

const NETWORK_CODES = ["FETCH_ERROR", "NETWORK_ERROR"];

const TIMEOUT_CODES = ["57014", "ETIMEDOUT", "ESOCKETTIMEDOUT"];

const PERMISSION_CODES = ["42501", "401", "403"];

const NOT_FOUND_CODES = ["PGRST116", "404"];

const VALIDATION_CODES = ["22P02", "23502", "23503", "23505"];

export function classifyError(error) {
  const code = String(error?.code || "").toUpperCase();
  const status = Number(error?.status || error?.statusCode);
  const message = String(error?.message || "").toLowerCase();

  if (
    TIMEOUT_CODES.includes(code) ||
    message.includes("timeout") ||
    message.includes("timed out")
  ) {
    return "TIMEOUT";
  }

  if (
    NETWORK_CODES.includes(code) ||
    message.includes("failed to fetch") ||
    message.includes("network request failed") ||
    message.includes("networkerror")
  ) {
    return "NETWORK";
  }

  if (
    PERMISSION_CODES.includes(code) ||
    status === 401 ||
    status === 403
  ) {
    return "PERMISSION";
  }

  if (
    NOT_FOUND_CODES.includes(code) ||
    status === 404
  ) {
    return "NOT_FOUND";
  }

  if (VALIDATION_CODES.includes(code)) {
    return "VALIDATION";
  }

  if (
    status >= 500 ||
    code.startsWith("08") ||
    code.startsWith("53")
  ) {
    return "SERVER";
  }

  if (code || error?.details || error?.hint) {
    return "DATABASE";
  }

  return "UNKNOWN";
}

import { useCallback, useState } from "react";
import MedicinesWorkspaceContext from "./MedicinesWorkspace";
export function MedicinesWorkspaceProvider({ children }) {
  const [history, setHistory] = useState([
    { type: "library" },
  ]);

  const current = history[history.length - 1];
  const navigateTo = useCallback((destination) => {
    setHistory((previous) => [...previous, destination]);
  }, []);

  const goBack = useCallback(() => {
    setHistory((previous) =>
      previous.length > 1
        ? previous.slice(0, -1)
        : previous,
    );
  }, []);

  const resetWorkspace = useCallback(() => {
    setHistory([{ type: "library" }]);
  }, []);

  const replaceWorkspace = useCallback((destination) => {
    setHistory([destination]);
  }, []);
  return (
    <MedicinesWorkspaceContext.Provider
  value={{
  history,
  current,
  navigateTo,
  goBack,
  resetWorkspace,
  replaceWorkspace,
}}
    >
      {children}
    </MedicinesWorkspaceContext.Provider>
  );
}
