import { useLayoutEffect } from "react";

import { getReadingState, saveReadingState } from "../services/readingState";
import {
  getScrollParent,
  getScrollTop,
  setScrollTop,
} from "../utils/scroll";

/* ========================================
   SCROLL MEMORY
   Restores where the student was on a page
   (the landing page, a set of search results)
   and keeps it up to date while they scroll.

   A new search has a new key, so it starts
   at the top.
======================================== */

export function useScrollMemory(ref, key, ready) {
  useLayoutEffect(() => {
    if (!ready || !key || !ref.current) return undefined;

    const scroller = getScrollParent(ref.current);
    const target = scroller === window ? window : scroller;

    setScrollTop(scroller, getReadingState(key)?.scroll?.page ?? 0);

    let timer = 0;

    const onScroll = () => {
      clearTimeout(timer);

      timer = setTimeout(() => {
        saveReadingState(key, {
          scroll: { page: getScrollTop(scroller) },
        });
      }, 200);
    };

    target.addEventListener("scroll", onScroll, { passive: true });

    return () => {
      target.removeEventListener("scroll", onScroll);
      clearTimeout(timer);
    };
  }, [ref, key, ready]);
}

/* ------------------------------------------------------------------
   Smart search
   ------------------------------------------------------------------
   One entry point for the whole search bar. The UI calls smartSearch()
   and renders whatever comes back. It never needs to know which
   Supabase function produced a result.

   Pipeline:
     query -> parse_search_query (subject + intent)
           -> search_medicines / search_drug_classes / search_interactions
           -> one normalised result object
------------------------------------------------------------------ */

// RxCT deliberately has no dosing content. Dosing errors can be fatal.
export const DOSAGE_NOTICE =
  "RxCT does not include dosing. Dosing errors can cause serious harm, " +
  "so check the BNF, the product label, or a pharmacist for doses.";

// A medicine counts as a confident match at or above this score.
// Below it, the UI should show "Did you mean" instead of jumping in.
const CONFIDENT_SCORE = 0.75;

// Which of the 5 medicine sections a search intent should open first.
// There is no dosing section, so "dosage" falls back to the overview.
export const INTENT_TO_SECTION = {
  overview: "overview",
  indications: "indications",
  adverse_effects: "adverse_effects",
  contraindications: "contraindications",
  interactions: "interactions",
  dosage: "overview",
};

export async function parseSearchQuery(q) {
  return executeRequest(
    "parseSearchQuery",
    async () => {
      const { data, error } = await supabase.rpc("parse_search_query", { q });

      if (error) throw error;
      return data?.[0] ?? null;
    },
    { query: q },
  );
}

export async function searchInteractions(q) {
  return executeRequest(
    "searchInteractions",
    async () => {
      const { data, error } = await supabase.rpc("search_interactions", { q });

      if (error) throw error;
      return data;
    },
    { query: q },
  );
}

// search_drug_classes returns one flat row per (class, medicine).
// This folds those rows into { type, title, slug, classes: [...] }.
export function groupClassRows(rows) {
  if (!rows || rows.length === 0) return null;

  const first = rows[0];
  const isGroup = first.result_type === "group";
  const byClass = new Map();

  for (const row of rows) {
    if (!byClass.has(row.drug_class)) {
      byClass.set(row.drug_class, []);
    }

    byClass.get(row.drug_class).push({
      id: row.medicine_id,
      name: row.medicine_name,
    });
  }

  return {
    type: isGroup ? "group" : "classes",
    title: isGroup ? first.group_name : null,
    slug: isGroup ? first.group_slug : null,
    classes: [...byClass].map(([drugClass, medicines]) => ({
      drugClass,
      medicines,
    })),
    medicineCount: rows.length,
  };
}

/*
  smartSearch(query) always resolves to this shape:

  {
    kind:         "medicines" | "classes" | "none",
    query:        what the user typed,
    subject:      the query with intent words stripped ("warfarin"),
    intent:       "overview" | "indications" | "adverse_effects" |
                  "contraindications" | "interactions" | "dosage",
    openSection:  which medicine section to open first,
    notice:       a message to show above results, or null,
    didYouMean:   true when the medicine matches are weak,
    medicines:    [{ id, name, drug_class, matched_term, score }],
    classResult:  output of groupClassRows(), or null,
    interactions: [{ medicine_id, medicine_name, drug_class, snippet }],
  }
*/
export async function smartSearch(rawQuery) {
  const query = (rawQuery ?? "").trim();

  const empty = {
    kind: "none",
    query,
    subject: "",
    intent: "overview",
    openSection: "overview",
    notice: null,
    didYouMean: false,
    medicines: [],
    classResult: null,
    interactions: [],
  };

  if (!query) return empty;

  const parsed = await parseSearchQuery(query);

  const subject = parsed?.subject ?? query;
  const intent = parsed?.intent ?? "overview";
  const wantsClass = parsed?.wants_class ?? false;

  const base = {
    ...empty,
    subject,
    intent,
    openSection: INTENT_TO_SECTION[intent] ?? "overview",
    notice: intent === "dosage" ? DOSAGE_NOTICE : null,
  };

  // Everything was an intent word ("dosage" on its own): nothing to search.
  if (!subject) return base;

  const [medicines, classRows, interactions] = await Promise.all([
    searchMedicines(query),
    searchDrugClasses(query),
    intent === "interactions" ? searchInteractions(query) : Promise.resolve([]),
  ]);

  const classResult = groupClassRows(classRows);
  const topScore = medicines[0]?.score ?? 0;
  const hasConfidentMedicine = topScore >= CONFIDENT_SCORE;

  let kind = "none";

  if (wantsClass && classResult) {
    kind = "classes";
  } else if (hasConfidentMedicine) {
    kind = "medicines";
  } else if (classResult) {
    kind = "classes";
  } else if (medicines.length > 0) {
    kind = "medicines";
  }

  return {
    ...base,
    kind,
    didYouMean: kind === "medicines" && !hasConfidentMedicine,
    medicines,
    classResult,
    interactions,
  };
}

const HISTORY_KEY = "rxct-medicine-history";
const PROGRESS_KEY = "rxct-study-progress";
const HISTORY_LIMIT = 30;

const readStorage = (key, fallback) => {
  try {
    const value = localStorage.getItem(key);
    return value ? JSON.parse(value) : fallback;
  } catch {
    return fallback;
  }
};

const writeStorage = (key, value) => {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch {
    // Storage may be unavailable or full.
  }
};

const same = (a, b) =>
  a.type === b.type &&
  String(a.name).toLowerCase() === String(b.name).toLowerCase();

export function getRecentActivity(limit = 5) {
  const history = readStorage(HISTORY_KEY, []);

  return Array.isArray(history) ? history.slice(0, limit) : [];
}

// Record that a medicine, class or group was opened.
export function recordVisit({ type, name }) {
  const item = { type, name };
  const history = readStorage(HISTORY_KEY, []);

  writeStorage(
    HISTORY_KEY,
    [
      { ...item, visitedAt: new Date().toISOString() },
      ...history.filter((entry) => !same(entry, item)),
    ].slice(0, HISTORY_LIMIT),
  );
}

export function getResumePoint() {
  const progress = readStorage(PROGRESS_KEY, null);

  if (!progress?.destination) return null;

  return {
    destination: progress.destination,
    title: progress.title || progress.name || "",
    section: progress.section || "",
  };
}

// The place to pick up from: where, and in which section.
export function saveResume({ destination, title, section }) {
  writeStorage(PROGRESS_KEY, { destination, title, section });
}

/* ========================================
   MEDICINE NOTES
   Notes written in Jotter. They are grouped
   by what the student was studying, so each
   medicine, class or group has one note
   collection that entries are added to,
   never a pile of duplicate records.

   {
     "medicine:amoxicillin": {
       type, name, className, groupName,
       entries: [{ id, text, section, createdAt }]
     }
   }

   Stored on the device. Resources reads it
   from here.
======================================== */

const NOTES_KEY = "rxct-medicine-notes-v1";
const DRAFT_KEY = "rxct-jotter-draft";

const read = (key, fallback) => {
  try {
    const value = localStorage.getItem(key);
    return value ? JSON.parse(value) : fallback;
  } catch {
    return fallback;
  }
};

const write = (key, value) => {
  try {
    localStorage.setItem(key, JSON.stringify(value));

    return true;
  } catch {
    return false;
  }
};

const entityKey = (context) =>
  context ? `${context.type}:${context.name.toLowerCase()}` : "general";

export function addNote(context, text) {
  const all = read(NOTES_KEY, {});
  const key = entityKey(context);

  const collection = all[key] || {
    type: context?.type || "general",
    name: context?.name || "General",
    className: context?.className || null,
    groupName: context?.groupName || null,
    entries: [],
  };

  collection.entries.unshift({
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    text,
    section: context?.sectionLabel || null,
    createdAt: new Date().toISOString(),
  });

  all[key] = collection;

  return write(NOTES_KEY, all);
}

export const getAllNotes = () => read(NOTES_KEY, {});

// An unfinished note survives closing Jotter or reloading the page.
export const getDraft = () => read(DRAFT_KEY, "");

export function saveDraft(text) {
  write(DRAFT_KEY, text);
}

export function clearDraft() {
  try {
    localStorage.removeItem(DRAFT_KEY);
  } catch {
    // Nothing to clear.
  }
}

import { supabase } from "./supabase";
import { executeRequest } from "./diagnostics/executeRequest";

/* ========================================
   LANDING PAGE DATA
   Classes (with how many medicines each
   holds and which group they belong to)
   and the therapeutic groups.
======================================== */

const FEATURED_LIMIT = 5;

const normalise = (value = "") => String(value).trim().toLowerCase();

export async function getLandingData() {
  return executeRequest("getLandingData", async () => {
    const [classResult, groupResult] = await Promise.all([
      supabase
        .from("drug_classes")
        .select("id, name, slug, medicines(name)")
        .order("name"),

      supabase
        .from("drug_class_groups")
        .select(
          "id, name, sort_order, drug_class_group_members(drug_class)",
        )
        .order("sort_order"),
    ]);

    if (classResult.error) throw classResult.error;
    if (groupResult.error) throw groupResult.error;

    // class name -> the first group that contains it
    const groupOfClass = new Map();

    for (const group of groupResult.data) {
      for (const member of group.drug_class_group_members || []) {
        const key = normalise(member.drug_class);

        if (!groupOfClass.has(key)) {
          groupOfClass.set(key, group.name);
        }
      }
    }

    const classes = classResult.data
      .map((row) => {
        const names = (row.medicines || [])
          .map((medicine) => medicine.name)
          .sort((a, b) => a.localeCompare(b));

        return {
          id: row.id,
          name: row.name,
          medicineCount: names.length,
          members: names.slice(0, 3),
          groupName: groupOfClass.get(normalise(row.name)) || "",
        };
      })
      .filter((item) => item.medicineCount > 0);

    return { classes };
  });
}

/* ========================================
   WHICH CLASSES TO FEATURE
   Restrained on purpose: classes the
   student explored recently come first,
   then the classes with the most medicines
   in the catalogue. Never more than five.
======================================== */

export function pickFeaturedClasses(classes, recentActivity = []) {
  const recentNames = recentActivity
    .filter((item) => item.type === "class")
    .map((item) => normalise(item.name));

  const recentRank = (item) => {
    const index = recentNames.indexOf(normalise(item.name));

    return index === -1 ? Infinity : index;
  };

  return [...classes]
    .sort((a, b) => {
      const byRecent = recentRank(a) - recentRank(b);

      // Infinity - Infinity is NaN, so compare explicitly
      if (recentRank(a) !== recentRank(b)) return byRecent < 0 ? -1 : 1;

      if (b.medicineCount !== a.medicineCount) {
        return b.medicineCount - a.medicineCount;
      }

      return a.name.localeCompare(b.name);
    })
    .slice(0, FEATURED_LIMIT);
}

let cache = null;
let timer = 0;

const load = () => {
  if (cache) return cache;

  try {
    cache = JSON.parse(localStorage.getItem(STORAGE_KEY)) || {};
  } catch {
    cache = {};

  return cache;

const persist = () => {
  const all = load();
  const keys = Object.keys(all);

  if (keys.length > MAX_ENTRIES) {
    keys
      .sort((a, b) => (all[a].updatedAt || 0) - (all[b].updatedAt || 0))
      .slice(0, keys.length - MAX_ENTRIES)
      .forEach((key) => delete all[key]);
  }

  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(all));
  } catch {
    // Storage may be unavailable or full.
  }
};

  clearTimeout(timer);
  timer = setTimeout(persist, SAVE_DELAY);
};

    scroll: { ...current.scroll, ...patch.scroll },
    expanded: { ...current.expanded, ...patch.expanded },

const NETWORK_CODES = ["FETCH_ERROR", "NETWORK_ERROR"];

const TIMEOUT_CODES = ["57014", "ETIMEDOUT", "ESOCKETTIMEDOUT"];

const PERMISSION_CODES = ["42501", "401", "403"];

const NOT_FOUND_CODES = ["PGRST116", "404"];

const VALIDATION_CODES = ["22P02", "23502", "23503", "23505"];

export function classifyError(error) {
  const code = String(error?.code || "").toUpperCase();
  const status = Number(error?.status || error?.statusCode);
  const message = String(error?.message || "").toLowerCase();

  if (
    TIMEOUT_CODES.includes(code) ||
    message.includes("timeout") ||
    message.includes("timed out")
  ) {
    return "TIMEOUT";
  }

  if (
    NETWORK_CODES.includes(code) ||
    message.includes("failed to fetch") ||
    message.includes("network request failed") ||
    message.includes("networkerror")
  ) {
    return "NETWORK";
  }

  if (
    PERMISSION_CODES.includes(code) ||
    status === 401 ||
    status === 403
  ) {
    return "PERMISSION";
  }

  if (
    NOT_FOUND_CODES.includes(code) ||
    status === 404
  ) {
    return "NOT_FOUND";
  }

  if (VALIDATION_CODES.includes(code)) {
    return "VALIDATION";
  }

  if (
    status >= 500 ||
    code.startsWith("08") ||
    code.startsWith("53")
  ) {
    return "SERVER";
  }

  if (code || error?.details || error?.hint) {
    return "DATABASE";
  }

  return "UNKNOWN";
}

const CACHE_KEY = "rxct-profile-cache-v1";
const CACHE_LIMIT = 12;

const readCache = () => {
  try {
    return JSON.parse(localStorage.getItem(CACHE_KEY)) || {};
  } catch {
    return {};
  }
};

const writeCache = (key, value) => {
  const all = readCache();

  all[key] = { value, at: Date.now() };

  Object.keys(all)
    .sort((a, b) => all[b].at - all[a].at)
    .slice(CACHE_LIMIT)
    .forEach((old) => delete all[old]);

  try {
    localStorage.setItem(CACHE_KEY, JSON.stringify(all));
  } catch {
    // Storage may be full. The profile still works online.
  }
};

async function withSavedCopy(key, load) {
  try {
    const value = await load();

    if (value) writeCache(key, value);

    return value;
  } catch (error) {
    const saved = readCache()[key];

    if (saved) return { ...saved.value, fromCache: true };

    throw error;
  }
}

const searchCache = new Map();

async function runSearch(query) {
  const key = query.trim().toLowerCase();

  if (!searchCache.has(key)) {
    searchCache.set(key, await smartSearch(query));
  }

// One clear winner opens its profile. Anything less clear shows a list.
const pickDirectMatch = (found) => {
  if (found.kind !== "medicines" || found.didYouMean) return null;

  const [top, second] = found.medicines;

  if (!top || top.score < DIRECT_MATCH_SCORE) return null;
  if (second && second.score >= DIRECT_MATCH_SCORE) return null;

  return top;
};

/* ========================================
   RESOLVE
   Turns whatever the student searched into
   one of: a medicine, a class, a group, or
   a list of search results.

   Explicit names always win. Search only
   runs when nothing matches exactly.
======================================== */

const PROFILE_LOADERS = {
  medicine: getMedicineProfile,
  class: getClassProfile,
  group: getGroupProfile,
};

const LOOKUPS = {
  medicine: lookupMedicine,
  class: lookupClass,
  group: lookupGroup,
};

export async function resolveQuery(query, kind) {
  const kinds = LOOKUPS[kind] ? [kind] : ["medicine", "class", "group"];

  const names = await Promise.all(
    kinds.map((type) => LOOKUPS[type](query).catch(() => null)),
  );

  const index = names.findIndex(Boolean);

  if (index !== -1) {
    const type = kinds[index];
    const data = await PROFILE_LOADERS[type](names[index]);

    if (data) return { type, name: data.name, data };
  }

  const found = await runSearch(query);
  const direct = pickDirectMatch(found);

  if (direct) {
    const data = await getMedicineProfile(direct.name);

    if (data) {
      const viaOtherName =
        direct.matched_term &&
        direct.matched_term.toLowerCase() !== direct.name.toLowerCase();

      return {
        type: "medicine",
        name: data.name,
        data,
        // Only an explicit request ("metronidazole adverse effects")
        // jumps to a section. Otherwise the full profile opens at the top.
        openSection: ["overview", "dosage"].includes(found.intent)
          ? ""
          : found.openSection,
        notice: found.notice,
        matchedNote: viaOtherName
          ? `Showing ${direct.name}, matched from "${direct.matched_term}".`
          : null,
      };
    }
  }

  return { type: "search", found };
}

// The importer writes these headings on their own line inside "overview".
const OVERVIEW_HEADINGS = {
  DESCRIPTION: "Description",
  "MECHANISM OF ACTION": "Mechanism of Action",
  "PHARMACOLOGICAL PROFILE": "Pharmacological Profile",
  PHARMACOKINETICS: "Pharmacokinetics",
};

const FOOTER_MARKER = "\n\nSource label:";
const SAFE_SOURCE_HOST = "https://dailymed.nlm.nih.gov/";

/* ========================================
   SENTENCE SPLITTING
======================================== */

const DOT = "\u2024";

const ABBREVIATIONS = [
  "e.g.",
  "i.e.",
  "vs.",
  "approx.",
  "dr.",
  "et al.",
  "u.s.",
  "inc.",
  "cf.",
  "fig.",
];

const escapeRegExp = (value) =>
  value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

const protectAbbreviations = (text) =>
  ABBREVIATIONS.reduce(
    (result, abbreviation) =>
      result.replace(
        new RegExp(escapeRegExp(abbreviation), "gi"),
        (match) => match.replaceAll(".", DOT),
      ),
    text,
  );

function splitSentences(text) {
  const pieces = protectAbbreviations(text).split(
    /([.!?])\s+(?=[A-Z0-9(])/,
  );

  const sentences = [];

  for (let index = 0; index < pieces.length; index += 2) {
    const sentence = pieces[index] + (pieces[index + 1] ?? "");

    sentences.push(sentence.replaceAll(DOT, ".").trim());
  }

  return sentences;
}

/* ========================================
   CLEANING
   Only removes things that are clearly
   navigation noise (cross references and
   section numbers). Clinical numbers are
   never touched.
======================================== */

export function cleanLabelText(raw) {
  return String(raw ?? "")
    .replace(/\s+/g, " ")
    // numbered sub-headings such as "4.1 Active Bleeding"
    .replace(/(^|[.!?)]\s+)\d{1,2}\.\d{1,2}\s+(?=[A-Z][a-z])/g, "$1")
    // padded cross references such as "( 4.1 )" ending an item
    .replace(
      /\s*\(\s+\d{1,2}(?:\.\d{1,2})?(?:\s*[,;]\s*\d{1,2}(?:\.\d{1,2})?)*\s+\)(?=\s+[A-Z])/g,
      ".",
    )
    // padded cross references inside a sentence
    .replace(
      /\s*\(\s+\d{1,2}(?:\.\d{1,2})?(?:\s*[,;]\s*\d{1,2}(?:\.\d{1,2})?)*\s+\)/g,
      " ",
    )
    // bracketed cross references such as "[see Warnings and Precautions]"
    .replace(/\[\s*see\b[^\]]*\]/gi, " ")
    // a leading numbered heading such as "4 CONTRAINDICATIONS"
    .replace(
      /^\s*\d{1,2}(?:\.\d{1,2})?\s+[A-Z][A-Z ,&/'-]{2,}?(?=\s+[A-Z][a-z])/,
      "",
    )
    .replace(/\s+([.,;:])/g, "$1")
    .replace(/\.(\s*\.)+/g, ".")
    .replace(/\s+/g, " ")
    .trim();
}

const comparable = (sentence) =>
  sentence.toLowerCase().replace(/[^a-z0-9]+/g, "");

// FDA labels repeat themselves: a short highlights line, then the same
// point again in full. Keep the longer version.
function dropRepeats(sentences) {
  const keys = sentences.map(comparable);

  return sentences.filter((sentence, index) => {
    const key = keys[index];

    if (key.length < 25) return true;

    return !keys.some((other, otherIndex) => {
      if (otherIndex === index) return false;
      if (other === key) return otherIndex < index;

      return other.length > key.length && other.includes(key);
    });
  });
}

export function toBullets(text) {
  const sentences = splitSentences(cleanLabelText(text)).filter(
    (sentence) => sentence.length >= 12,
  );

  return dropRepeats(sentences);
}

/* ========================================
   SECTION PARSING
======================================== */

function formatEffectiveDate(value) {
  const match = String(value).match(/^(\d{4})(\d{2})(\d{2})$/);

  if (!match) return value;

  const date = new Date(
    Number(match[1]),
    Number(match[2]) - 1,
    Number(match[3]),
  );

  if (Number.isNaN(date.getTime())) return value;

  return date.toLocaleDateString(undefined, {
    day: "numeric",
    month: "short",
    year: "numeric",
  });
}

export function splitSourceFooter(body) {
  const text = String(body ?? "");
  const index = text.lastIndexOf(FOOTER_MARKER);

  if (index === -1) return { text, source: null };

  const footer = text.slice(index).trim();

  const match = footer.match(
    /^Source label:\s*(.+?),\s*US FDA prescribing information,\s*effective\s+(.+?)\.\s+(https?:\/\/\S+)$/,
  );

  const source = match
    ? {
        title: match[1],
        effective: formatEffectiveDate(match[2]),
        // Only link to DailyMed. Never trust other hosts from stored text.
        url: match[3].startsWith(SAFE_SOURCE_HOST) ? match[3] : null,
      }
    : null;

  return { text: text.slice(0, index), source };
}

function splitBlocks(text) {
  const blocks = [];
  let current = { title: null, lines: [] };

  for (const line of text.split("\n")) {
    const heading = OVERVIEW_HEADINGS[line.trim()];

    if (heading) {
      blocks.push(current);
      current = { title: heading, lines: [] };
    } else {
      current.lines.push(line);
    }
  }

  blocks.push(current);

  return blocks.map((block) => ({
    title: block.title,
    text: block.lines.join(" "),
  }));
}

/*
  Returns:
  {
    blocks:    [{ title: string | null, bullets: string[] }],
    source:    { title, effective, url } | null,
    truncated: true when the importer shortened the text,
  }
*/
export function parseSectionBody(body) {
  const { text, source } = splitSourceFooter(body);

  const truncated = /\[Shortened here\./.test(text);
  const withoutMarker = text.replace(/\[Shortened here\.[^\]]*\]/, "");

  const blocks = splitBlocks(withoutMarker)
    .map((block) => ({
      title: block.title,
      bullets: toBullets(block.text),
    }))
    .filter((block) => block.bullets.length > 0);

  return { blocks, source, truncated };
}

// A few short, readable bullets for the "At a Glance" panel.
export function keyPoints(bullets, max = 3) {
  const concise = bullets.filter(
    (bullet) => bullet.length >= 20 && bullet.length <= 260,
  );

  return (concise.length > 0 ? concise : bullets)
    .slice(0, max)
    .map((bullet) =>
      bullet.length > 260
        ? bullet.slice(0, 257).trimEnd() + "\u2026"
        : bullet,
    );
}

/* ========================================
   NAMES
   Drug class names arrive in mixed casing
   ("Azole antifungal", "NSAID", "alpha-
   Adrenergic Blocker"). Capitalise each
   word, but never break existing acronyms.
======================================== */

const SMALL_WORDS = new Set([
  "of",
  "and",
  "the",
  "in",
  "for",
  "with",
  "a",
  "an",
  "to",
  "or",
]);

export function formatClassName(name) {
  if (!name) return "";

  return String(name)
    .split(" ")
    .map((word, index) => {
      const match = word.match(/^([^A-Za-z]*)(.*)$/);
      const prefix = match[1];
      const rest = match[2];

      if (!rest) return word;

      if (index > 0 && SMALL_WORDS.has(rest.toLowerCase())) {
        return prefix + rest.toLowerCase();
      }

      const hasInnerCapitals = /[A-Z]/.test(rest.slice(1));
      const body = hasInnerCapitals ? rest : rest.toLowerCase();

      return prefix + body.charAt(0).toUpperCase() + body.slice(1);
    })
    .join(" ");
}

/* ========================================
   SCROLL HELPERS
   RxCT does not assume whether the window
   or a layout container scrolls, so these
   find the real scroller at runtime.
======================================== */

export function getScrollParent(node) {
  let element = node?.parentElement;

  while (
    element &&
    element !== document.body &&
    element !== document.documentElement
  ) {
    const { overflowY } = getComputedStyle(element);

    if (
      /(auto|scroll|overlay)/.test(overflowY) &&
      element.scrollHeight > element.clientHeight + 1
    ) {
      return element;
    }

    element = element.parentElement;
  }

  return window;
}

export const getScrollTop = (scroller) =>
  scroller === window ? window.scrollY : scroller.scrollTop;

export function setScrollTop(scroller, value) {
  if (scroller === window) {
    window.scrollTo({ top: value, left: 0, behavior: "auto" });
  } else {
    scroller.scrollTop = value;
  }
}

// Distance of a node from the top of the scrollable content.
export function offsetWithin(node, scroller) {
  const base =
    scroller === window ? 0 : scroller.getBoundingClientRect().top;

  return (
    node.getBoundingClientRect().top - base + getScrollTop(scroller)
  );
}

// Where, in the viewport, sticky UI must sit so it clears the Topbar
// and stays inside the scrolling area. Works whether the window or a
// layout container scrolls, and whether the Topbar overlaps or not.
export function measureViewportTop(node) {
  const scroller = getScrollParent(node);

  const bar =
    document.querySelector("[data-global-search]")?.closest("header") ||
    document.querySelector("header");

  let top = scroller === window ? 0 : scroller.getBoundingClientRect().top;

  if (bar) {
    const { position } = getComputedStyle(bar);

    if (position === "sticky" || position === "fixed") {
      top = Math.max(top, bar.getBoundingClientRect().bottom);
    }
  }

  return Math.max(0, Math.round(top));
}

export function scrollToY(scroller, value, smooth) {
  const behavior =
    smooth && !window.matchMedia?.("(prefers-reduced-motion: reduce)").matches
      ? "smooth"
      : "auto";

  if (scroller === window) {
    window.scrollTo({ top: value, left: 0, behavior });
  } else {
    scroller.scrollTo({ top: value, left: 0, behavior });
  }
}

/* ========================================
   SCIENTIFIC NOTATION
   Display-only conversion of plain-text
   notation into proper typography:
   H2O -> H₂O, Ca2+ -> Ca²⁺, beta- -> β-.

   It never touches doses, strengths or any
   other clinical number, and the stored
   text is never modified.

   Complex equations will use KaTeX once
   the content actually contains them.
======================================== */

const SUBSCRIPT = {
  0: "₀", 1: "₁", 2: "₂", 3: "₃", 4: "₄",
  5: "₅", 6: "₆", 7: "₇", 8: "₈", 9: "₉",
};

const GREEK = {
  alpha: "α",
  beta: "β",
  gamma: "γ",
  delta: "δ",
  kappa: "κ",
  mu: "μ",
};

export function toScientific(text) {
  return String(text ?? "")
    .replace(/\bH2O2\b/g, "H₂O₂")
    .replace(/\bH2O\b/g, "H₂O")
    .replace(/\bCO2\b/g, "CO₂")
    .replace(/\bN2O\b/g, "N₂O")
    .replace(/\bNH4\+/g, "NH₄⁺")
    .replace(/\b(Ca|Mg|Fe|Zn|Cu)2\+/g, "$1²⁺")
    .replace(/\b(Na|K|Li)\+(?!\w)/g, "$1⁺")
    .replace(
      /\bH2(?=[\s-]+(?:[Rr]eceptor|[Bb]locker|[Aa]ntagonist))/g,
      "H₂",
    )
    .replace(
      /\b5-HT([1-7])(?!\d)/g,
      (_, digit) => "5-HT" + SUBSCRIPT[digit],
    )
    .replace(
      /\b(alpha|beta|gamma|delta|kappa|mu)-(?=[A-Za-z0-9])/gi,
      (_, name) => GREEK[name.toLowerCase()] + "-",
    );
}

export const REVIEW_LABELS = {
  imported: "Imported text, not yet reviewed by RxCT.",
  reviewed: "Reviewed by RxCT.",
};

const slug = (value) =>
  String(value)
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");

export function parseEditorial(body) {
  const blocks = [{ title: null, nodes: [] }];
  let paragraph = [];
  let list = null;

  const last = () => blocks[blocks.length - 1];

  const flush = () => {
    if (paragraph.length) {
      last().nodes.push({ type: "p", text: paragraph.join(" ") });
      paragraph = [];
    }

    if (list) {
      last().nodes.push({ type: "ul", items: list });
      list = null;
    }
  };

  for (const raw of String(body ?? "").split("\n")) {
    const line = raw.trim();

    if (!line) {
      flush();
    } else if (line.startsWith("## ")) {
      flush();
      blocks.push({ title: line.slice(3).trim(), nodes: [] });
    } else if (line.startsWith("- ")) {
      if (paragraph.length) flush();
      (list ??= []).push(line.slice(2).trim());
    } else {
      if (list) flush();
      paragraph.push(line);
    }
  }

  flush();

  return blocks.filter((block) => block.title || block.nodes.length);
}

      if (!keys.includes(key) || key === active) return;

      const scroller = getScrollParent(rootRef.current);

      positions.current[active] = getScrollTop(scroller);

      saveReadingState(stateKey, {
        section: key,
        scroll: {
          [active]: positions.current[active],
        },
      });

      setActive(key);
    },
    [active, keys, stateKey],
  );

  /* ========================================
     STICKY BAR POSITION
  ======================================== */

  const measure = useCallback(() => {
    const slot = slotRef.current;

    if (!slot) return;

    const top = viewportTop.current;
    const rect = slot.getBoundingClientRect();
    const stuck = rect.top <= top;

    setBar((state) =>
      state.stuck === stuck &&
      state.top === top &&
      state.left === rect.left &&
      state.width === rect.width
        ? state
        : {
            stuck,
            top,
            left: rect.left,
            width: rect.width,
          },
    );
  }, []);

  /* ========================================
     PLACEMENT
  ======================================== */

  useLayoutEffect(() => {
    activeRef.current = active;
  }, [active]);

  useLayoutEffect(() => {
    const root = rootRef.current;

    if (!root) return;

    const scroller = getScrollParent(root);
    const stored = positions.current[active];

    viewportTop.current = measureViewportTop(root);

    if (firstRun.current) {
      firstRun.current = false;

      if (initialSub) {
        const element = root.querySelector(
          `[data-sub="${CSS.escape(
            `${active}.${initialSub}`,
          )}"]`,
        );

        if (element) {
          setScrollTop(scroller, positionFor(element));
          measure();
          return;
        }
      }

      if (keys.includes(initialSection)) {
        if (
          stored != null &&
          saved?.section === initialSection
        ) {
          setScrollTop(scroller, stored);
        } else {
          setScrollTop(
            scroller,
            positionFor(panelRef.current),
          );
        }
      } else {
        setScrollTop(scroller, stored ?? 0);
      }

      measure();
      return;
    }

    if (stored != null) {
      setScrollTop(scroller, stored);
    } else {
      const target = positionFor(panelRef.current);

      if (getScrollTop(scroller) > target) {
        setScrollTop(scroller, target);
      }
    }

    measure();

    // Section definitions remain fixed for this profile.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [active]);

  /* ========================================
     SCROLL AND RESIZE
  ======================================== */

  useEffect(() => {
    let frame = 0;
    let timer = 0;

    const onScroll = () => {
      cancelAnimationFrame(frame);

      frame = requestAnimationFrame(() => {
        measure();

        const section = activeRef.current;

        positions.current[section] = getScrollTop(
          getScrollParent(rootRef.current),
        );

        clearTimeout(timer);

        timer = setTimeout(() => {
          saveReadingState(stateKey, {
            section,
            scroll: {
              [section]: positions.current[section],
            },
          });
        }, 150);
      });
    };

    const onResize = () => {
      viewportTop.current = measureViewportTop(
        rootRef.current,
      );

      onScroll();
    };

    window.addEventListener("scroll", onScroll, {
      capture: true,
      passive: true,
    });

    window.addEventListener("resize", onResize);

    return () => {
      window.removeEventListener("scroll", onScroll, {
        capture: true,
      });

      window.removeEventListener("resize", onResize);

      cancelAnimationFrame(frame);
      clearTimeout(timer);
    };
  }, [measure, stateKey]);

  /* ========================================
     KEEP ACTIVE TAB IN VIEW
  ======================================== */

  useEffect(() => {
    const tabs = tabsRef.current;

    const button = tabs?.querySelector(
      '[aria-current="true"]',
    );

    if (!tabs || !button) return;

    tabs.scrollTo({
      left:
        button.offsetLeft -
        (tabs.clientWidth - button.offsetWidth) / 2,
      behavior: "auto",
    });
  }, [active, bar.stuck]);

  /* ========================================
     REPORTING
  ======================================== */

  const onSectionChangeRef = useRef(onSectionChange);

  useLayoutEffect(() => {
    onSectionChangeRef.current = onSectionChange;
  });

  useEffect(() => {
    saveReadingState(stateKey, {
      section: active,
    });

    onSectionChangeRef.current?.(
      active,
      current?.label || "",
    );
  }, [active, current?.label, stateKey]);

  const contextKey = JSON.stringify(context || null);

  useEffect(() => {
    setStudyContext(
      context
        ? {
            ...context,
            section: active,
            sectionLabel: current?.label,
          }
        : null,
    );

    return () => setStudyContext(null);

    // contextKey represents the context object.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [active, current?.label, contextKey]);

  /* ========================================
     RENDER
  ======================================== */
const api = {
  isOpen,
  toggle,
};

  return (
    <div className={styles.profile} ref={rootRef}>
      <button
        type="button"
        className={styles.back}
        onClick={onBack}
      >
        <i
          className="fa-solid fa-arrow-left"
          aria-hidden="true"
        />
        Back
      </button>

      <header className={styles.header}>
        {kindLabel && (
          <span className={styles.eyebrow}>
            {kindLabel}
          </span>
        )}

        <h1>{title}</h1>

        {subtitle && <p>{subtitle}</p>}

        {trail.length > 0 && (
          <nav
            className={styles.trail}
            aria-label="Classification"
          >
            {trail.map((item) => (
              <button
                type="button"
                key={item.label}
                onClick={item.onClick}
              >
                {item.label}
              </button>
            ))}
          </nav>
        )}
      </header>

      {/* STICKY SECTION NAVIGATION */}

      {sections.length > 1 && (
        <div
          className={styles.slot}
          ref={slotRef}
        >
          <nav
            className={
              bar.stuck
                ? `${styles.bar} ${styles.barStuck}`
                : styles.bar
            }
            style={
              bar.stuck
                ? {
                    top: bar.top,
                    left: bar.left,
                    width: bar.width,
                  }
                : undefined
            }
            aria-label="Study sections"
          >
            {bar.stuck && (
              <p className={styles.barTitle}>
                {title}
              </p>
            )}

            <div
              className={styles.tabs}
              ref={tabsRef}
            >
              {sections.map((section) => (
                <button
                  type="button"
                  key={section.key}
                  title={section.label}
                  aria-label={section.label}
                  aria-current={
                    active === section.key
                      ? "true"
                      : undefined
                  }
                  className={
                    active === section.key
                      ? `${styles.tab} ${styles.tabActive}`
                      : styles.tab
                  }
                  onClick={() => select(section.key)}
                >
                  {section.icon && (
                    <i
                      className={`fa-solid ${section.icon}`}
                      aria-hidden="true"
                    />
                  )}

                  <span className={styles.tabLabel}>
                    {section.short || section.label}
                  </span>
                </button>
              ))}
            </div>
          </nav>
        </div>
      )}

      {notices.map((notice) => (
        <p
          className={styles.notice}
          role="note"
          key={notice}
        >
          {notice}
        </p>
      ))}

      {glance && (
        <aside
          className={styles.glance}
          aria-label="At a glance"
        >
          <h2 className={styles.glanceTitle}>
            <button
              type="button"
              aria-expanded={isOpen("glance")}
              onClick={() => toggle("glance")}
            >
              <i
                className="fa-solid fa-eye"
                aria-hidden="true"
              />

              <span>At a glance</span>

              <span
                className={styles.chevron}
                aria-hidden="true"
              />
            </button>
          </h2>

          {isOpen("glance") && (
            <div className={styles.glanceBody}>
          {glance({ ...api, jump: select })}
            </div>
          )}
        </aside>
      )}

      <article
        key={active}
        className={styles.card}
        ref={panelRef}
        aria-labelledby={`heading-${active}`}
      >
        <h2 id={`heading-${active}`}>
          {current?.label}
        </h2>

        {current?.render(api)}

        {(previous || next) && (
          <nav
            className={styles.pager}
            aria-label="Neighbouring sections"
          >
            {previous ? (
              <button
                type="button"
                onClick={() => select(previous.key)}
              >
                <i
                  className="fa-solid fa-arrow-left"
                  aria-hidden="true"
                />

                {previous.label}
              </button>
            ) : (
              <span />
            )}

            {next && (
              <button
                type="button"
                onClick={() => select(next.key)}
              >
                {next.label}

                <i
                  className="fa-solid fa-arrow-right"
                  aria-hidden="true"
                />
              </button>
            )}
          </nav>
        )}
      </article>
    </div>
  );
}

export default StudyProfile;

/* ========================================

   The broadest level: what the group covers,
   which classes belong to it and how they
   relate therapeutically.

const GROUP_SECTIONS = [
  { key: "overview", label: "Overview", short: "Overview", icon: "fa-circle-info" },
  {
    key: "classification",
    label: "Classification",
    short: "Classification",
    icon: "fa-sitemap",
  },
  {
    key: "mechanisms_of_action",
    label: "Mechanisms of Action",
    short: "Mechanisms",
    icon: "fa-flask",
  },
  {
    key: "therapeutic_applications",
    label: "Therapeutic Applications",
    short: "Applications",
    icon: "fa-stethoscope",
  },
  {
    key: "important_considerations",
    label: "Important Considerations",
    short: "Considerations",
    icon: "fa-triangle-exclamation",
  },
];

function GroupProfile({
  group,
  initialSection,
  initialSub,
  onBack,
  onOpenClass,
  onSectionChange,
}) {
  const count = group.classes.length;

  const renderContent = (key) => (ctx) => {
    const record = group.content.find((item) => item.section === key);

    const classList =
      key === "classification" && count > 0 ? (
        <Subsection
          id="classification.classes"
          title="Drug classes in this group"
          open={ctx.isOpen("classification.classes")}
          onToggle={() => ctx.toggle("classification.classes")}
        >
          <LinkList
            items={group.classes.map((item) => ({
              key: item.id,
              kind: "class",
              label: formatClassName(item.name),
              meta: `${item.medicineCount} ${
                item.medicineCount === 1 ? "medicine" : "medicines"
              }`,
              onClick: () => onOpenClass(item.name),
            }))}
          />
        </Subsection>
      ) : null;

    if (!record?.body?.trim()) {
      return classList || (
        <EmptySection>
          Study content for this section has not been added yet.
        </EmptySection>
      );
    }

    return (
      <>
        <SourceLine
          source={record.sources?.name}
          status={record.review_status}
        />

        <EditorialBody body={record.body} section={key} ctx={ctx} />

        {classList}
      </>
    );
  };

  const sections = GROUP_SECTIONS.map(({ key, label, short, icon }) => ({
    key,
    label,
    short,
    icon,
    render: renderContent(key),
  }));

  const overview = group.content.find(
    (item) => item.section === "overview",
  )?.body;

  const brief = parseEditorial(overview)
    .flatMap((block) => block.nodes)
    .find((node) => node.type === "p")?.text;

  const renderGlance = ({ jump }) => (
    <GlanceBlock title="In brief">
      <p className={styles.glanceText}>{brief}</p>

      <GlanceLink onClick={() => jump("overview")}>
        Read the overview
      </GlanceLink>
    </GlanceBlock>
  );

  return (
    <StudyProfile
      stateKey={`group:${group.name.toLowerCase()}`}
      kindLabel="THERAPEUTIC GROUP"
      title={group.name}
      subtitle={`Therapeutic group with ${count} drug ${
        count === 1 ? "class" : "classes"
      }`}
      notices={
        group.fromCache
          ? ["Showing a saved copy. Reconnect to check for updates."]
          : []
      }
      glance={brief ? renderGlance : undefined}
      sections={sections}
      initialSection={initialSection}
      initialSub={initialSub}
      context={{
        type: "group",
        name: group.name,
        className: null,
        groupName: group.name,
      }}
      onBack={onBack}
      onSectionChange={onSectionChange}
    />
  );
}

export default GroupProfile;

/* ========================================
   OTHER LABELS THAT MENTION THIS MEDICINE
   A text match, not a verified interaction
   check. Loaded the first time it is opened.
======================================== */

const mentionsCache = new Map();

function MentionedLabels({ name, onOpenMedicine }) {
  const [state, setState] = useState(
    mentionsCache.has(name)
      ? { status: "ready", rows: mentionsCache.get(name) }
      : { status: "loading", rows: [] },
  );

  useEffect(() => {
    if (mentionsCache.has(name)) return undefined;

    let active = true;

    searchInteractions(name)
      .then((rows) => {
        mentionsCache.set(name, rows || []);

        if (active) setState({ status: "ready", rows: rows || [] });
      })
      .catch(() => {
        if (active) setState({ status: "error", rows: [] });
      });

    return () => {
      active = false;
    };
  }, [name]);

  if (state.status === "loading") {
    return <p className={styles.muted}>Checking other labels...</p>;
  }

  if (state.status === "error") {
    return (
      <p className={styles.muted}>
        Other labels could not be checked right now.
      </p>
    );
  }

  if (state.rows.length === 0) {
    return (
      <p className={styles.muted}>
        No other label in RxCT mentions {name}.
      </p>
    );
  }

  return (
    <>
      <p className={styles.muted}>
        Medicines whose interaction text mentions {name}. This is a text
        match, not a verified interaction check.
      </p>

      <LinkList
        items={state.rows.slice(0, 12).map((row) => ({
          key: row.medicine_id,
          kind: "medicine",
          label: row.medicine_name,
          meta: formatClassName(row.drug_class),
          onClick: () => onOpenMedicine(row.medicine_name),
        }))}
      />
    </>
  );
}

   Five sections. Overview carries the

function MedicineProfile({
  medicine,
  initialSection,
  initialSub,
  notice,
  matchedNote,
  onBack,
  onOpenClass,
  onOpenGroup,
  onOpenMedicine,
  onSectionChange,
}) {
  // A medicine can have the same section from more than one source.
  // Prefer a reviewed row, otherwise take the first.
  const records = useMemo(() => {
    const bySection = {};

    for (const record of medicine.medicine_sections || []) {
      const existing = bySection[record.section];

      if (
        !existing ||
        (record.review_status === "reviewed" &&
          existing.review_status !== "reviewed")
      ) {
        bySection[record.section] = record;
      }
    }

    return bySection;
  }, [medicine]);

  const parsed = useMemo(() => {
    const result = {};

    for (const [key, record] of Object.entries(records)) {
      result[key] = parseSectionBody(record.body);
    }

    return result;
  }, [records]);

  const className = formatClassName(medicine.drug_class);

  /* ========================================
     SECTION RENDERERS
  ======================================== */

  const sub = (ctx, id, title, content) => (
    <Subsection
      key={id}
      id={id}
      title={title}
      open={ctx.isOpen(id)}
      onToggle={() => ctx.toggle(id)}
    >
      {content}
    </Subsection>
  );

  const renderOverview = (ctx) => {
    const blocks = parsed.overview?.blocks || [];
    const byTitle = (title) => blocks.find((block) => block.title === title);

    const description =
      byTitle("Description") || blocks.find((block) => !block.title);

    const bullets = (id, block) => (
      <Bullets id={id} items={block.bullets} ctx={ctx} />
    );

    const mechanism = byTitle("Mechanism of Action");
    const kinetics = byTitle("Pharmacokinetics");
    const profile = byTitle("Pharmacological Profile");

    return (
      <>
        {description &&
          sub(ctx, "overview.description", "Description",
            bullets("overview.description", description))}

        {mechanism &&
          sub(ctx, "overview.mechanism", "Mechanism of action",
            bullets("overview.mechanism", mechanism))}

        {kinetics &&
          sub(ctx, "overview.pharmacokinetics", "Pharmacokinetics",
            bullets("overview.pharmacokinetics", kinetics))}

        {profile &&
          sub(ctx, "overview.profile", "Other pharmacological properties",
            bullets("overview.profile", profile))}
      </>
    );
  };

  const renderBlocks = (key, ctx) => {
    const blocks = parsed[key]?.blocks || [];

    if (blocks.length === 1 && !blocks[0].title) {
      return (
        <Bullets
          id={key}
          items={blocks[0].bullets}
          ctx={ctx}
          tone={key === "contraindications" ? "warning" : undefined}
        />
      );
    }

    return blocks.map((block, index) => {
      const id = `${key}.${index}`;

      return sub(
        ctx,
        id,
        block.title || "General",
        <Bullets
          id={id}
          items={block.bullets}
          ctx={ctx}
          tone={key === "contraindications" ? "warning" : undefined}
        />,
      );
    });
  };

  const renderSection = (key) => (ctx) => {
    const record = records[key];
    const data = parsed[key];

    if (!record || !data || data.blocks.length === 0) {
      return (
        <EmptySection>
          No {SECTION_LABELS[key].toLowerCase()} information is recorded
          for {medicine.name} yet.
        </EmptySection>
      );
    }

    return (
      <>
        <SourceLine
          source={record.sources?.name}
          status={record.review_status}
        />

        {key === "overview"
          ? renderOverview(ctx)
          : renderBlocks(key, ctx)}

        {key === "interactions" &&
          sub(
            {
              isOpen: (id) => ctx.isOpen(id, false),
              toggle: (id) => ctx.toggle(id, false),
            },
            "interactions.mentions",
            "Mentioned in other labels",
            ctx.isOpen("interactions.mentions", false) && (
              <MentionedLabels
                name={medicine.name}
                onOpenMedicine={onOpenMedicine}
              />
            ),
          )}

        <SourceFooter
          source={data.source}
          truncated={data.truncated}
        />
      </>
    );
  };

  const hasContent = (key) => parsed[key]?.blocks.length > 0;

  const sections = SECTION_ORDER.filter(hasContent).map((key) => ({
    key,
    label: SECTION_LABELS[key],
    ...SECTION_META[key],
    render: renderSection(key),
  }));

  /* ========================================
     AT A GLANCE
     Classification first, then the points a
     student most needs to see. Full text
     stays in the sections.
  ======================================== */

  const points = (key) =>
    (parsed[key]?.blocks || [])
      .flatMap((block) => block.bullets)
      .filter((text) => text.length >= 20 && text.length <= 220)
      .slice(0, 3);

  const contraindications = points("contraindications");
  const interactions = points("interactions");

  const renderGlance = ({ jump }) => (
    <>
      {(medicine.group || medicine.drug_class) && (
        <GlanceBlock title="Classification">
          <LinkList
            items={[
              medicine.group && {
                key: "group",
                kind: "group",
                label: medicine.group.name,
                meta: "Therapeutic group",
                onClick: () => onOpenGroup(medicine.group.name),
              },
              medicine.drug_class && {
                key: "class",
                kind: "class",
                label: className,
                meta: "Drug class",
                onClick: () => onOpenClass(medicine.drug_class),
              },
            ].filter(Boolean)}
          />
        </GlanceBlock>
      )}

      {contraindications.length > 0 && (
        <GlanceBlock title="Key contraindications">
          <GlancePoints items={contraindications} tone="warning" />

          <GlanceLink onClick={() => jump("contraindications")}>
            Read all contraindications
          </GlanceLink>
        </GlanceBlock>
      )}

      {interactions.length > 0 && (
        <GlanceBlock title="Key interactions">
          <GlancePoints items={interactions} />

          <GlanceLink onClick={() => jump("interactions")}>
            Read all interactions
          </GlanceLink>
        </GlanceBlock>
      )}
    </>
  );

  /* ========================================
     RENDER
  ======================================== */

  const trail = [
    medicine.group && {
      label: medicine.group.name,
      onClick: () => onOpenGroup(medicine.group.name),
    },
    medicine.drug_class && {
      label: className,
      onClick: () => onOpenClass(medicine.drug_class),
    },
  ].filter(Boolean);

  return (
    <StudyProfile
      stateKey={`medicine:${medicine.name.toLowerCase()}`}
      kindLabel="MEDICINE"
      title={medicine.name}
      subtitle={className || "Drug class not recorded"}
      trail={trail}
      notices={[
        notice,
        matchedNote,
        medicine.fromCache &&
          "Showing a saved copy. Reconnect to check for updates.",
      ].filter(Boolean)}
      glance={renderGlance}
      sections={sections}
      initialSection={initialSection}
      initialSub={initialSub}
      context={{
        type: "medicine",
        name: medicine.name,
        className: className || null,
        groupName: medicine.group?.name || null,
      }}
      onBack={onBack}
      onSectionChange={onSectionChange}
    />
  );
}

import { useEffect, useRef, useState, useSyncExternalStore } from "react";

import {
  getStudyContext,
  subscribeStudyContext,
} from "../../services/studyContext";
import { addNote, getDraft, saveDraft, clearDraft } from "../../services/notes";

import styles from "./Jotter.module.css";

/* ========================================
   JOTTER
   A floating note pad for the Medicines
   screens. It picks up what the student is
   studying (medicine, class or group, and
   the section in view) so the note is filed
   against it.

   Ctrl/Cmd + Shift + J opens and closes it.
======================================== */

function Jotter() {
  const context = useSyncExternalStore(
    subscribeStudyContext,
    getStudyContext,
  );

  const [open, setOpen] = useState(false);
  const [text, setText] = useState(getDraft);
  const [saved, setSaved] = useState(false);

  const textareaRef = useRef(null);

  /* ========================================
     SHORTCUT AND ESCAPE
  ======================================== */

  useEffect(() => {
    const onKeyDown = (event) => {
      const isShortcut =
        (event.ctrlKey || event.metaKey) &&
        event.shiftKey &&
        event.key.toLowerCase() === "j";

      if (isShortcut) {
        const target = event.target;

        const typingElsewhere =
          target !== textareaRef.current &&
          (target?.tagName === "INPUT" ||
            target?.tagName === "TEXTAREA" ||
            target?.isContentEditable);

        if (typingElsewhere) return;

        event.preventDefault();
        setOpen((value) => !value);

        return;
      }

      if (event.key === "Escape") setOpen(false);
    };

    window.addEventListener("keydown", onKeyDown);

    return () => window.removeEventListener("keydown", onKeyDown);
  }, []);

  useEffect(() => {
    if (open) textareaRef.current?.focus();
  }, [open]);

  useEffect(() => {
    if (!saved) return undefined;

    const timer = setTimeout(() => setSaved(false), 2500);

    return () => clearTimeout(timer);
  }, [saved]);

  /* ========================================
     ACTIONS
  ======================================== */

  const onChange = (event) => {
    setText(event.target.value);
    saveDraft(event.target.value);
  };

  const save = () => {
    const note = text.trim();

    if (!note) return;

    if (addNote(context, note)) {
      clearDraft();
      setText("");
      setOpen(false);
      setSaved(true);
    }
  };

  const where = context
    ? [context.name, context.sectionLabel].filter(Boolean).join(", ")
    : "General note";

  /* ========================================
     RENDER
  ======================================== */

  return (
    <>
      {open && (
        <div className={styles.panel} role="dialog" aria-label="Jotter">
          <div className={styles.header}>
            <div>
              <h2>Jotter</h2>
              <p>{where}</p>
            </div>

            <button
              type="button"
              className={styles.close}
              aria-label="Close Jotter"
              onClick={() => setOpen(false)}
            >
              <i className="fa-solid fa-xmark" aria-hidden="true"></i>
            </button>
          </div>

          <textarea
            ref={textareaRef}
            value={text}
            onChange={onChange}
            placeholder="Write a quick note..."
            rows={5}
          />

          <div className={styles.actions}>
            <button
              type="button"
              className={styles.save}
              disabled={!text.trim()}
              onClick={save}
            >
              Save note
            </button>
          </div>
        </div>
      )}

      {saved && (
        <p className={styles.toast} role="status">
          Note saved
        </p>
      )}

      <button
        type="button"
        className={styles.button}
        aria-label="Open Jotter"
        aria-expanded={open}
        onClick={() => setOpen((value) => !value)}
      >
        <i
          className={open ? "fa-solid fa-xmark" : "fa-solid fa-pen"}
          aria-hidden="true"
        ></i>
      </button>
    </>
  );
}

  ======================================== */

  const selectTab = (key) => {
    setActiveTab(key);

    panelRef.current?.scrollIntoView({
      behavior: "smooth",
      block: "start",
    });
  };

  const toggleBlock = (key) => {
    setExpandedBlocks((previous) => ({
      ...previous,
      [key]: !previous[key],
    }));
  };

  /* ========================================
     NO STUDY CONTENT
  ======================================== */

  if (availableTabs.length === 0) {
    return (
      <div className={styles.empty}>
        <h3>Study material unavailable</h3>
        <p>
          There is currently no recorded study content for this
          medicine.
        </p>
      </div>
    );
  }

  const record = records[activeTab];
  const current = parsed[activeTab];

  /* ========================================
     RENDER HELPERS
  ======================================== */

  const renderBlock = (block, index) => {
    const key = `${activeTab}:${index}`;
    const expanded = Boolean(expandedBlocks[key]);
    const collapsible = block.bullets.length > VISIBLE_BULLETS + 2;

    const visible =
      collapsible && !expanded
        ? block.bullets.slice(0, VISIBLE_BULLETS)
        : block.bullets;

    return (
      <section className={styles.block} key={key}>
        {block.title && <h3 className={styles.blockTitle}>{block.title}</h3>}

        <ul
          className={
            activeTab === "contraindications"
              ? `${styles.bulletList} ${styles.dangerList}`
              : styles.bulletList
          }
        >
          {visible.map((bullet, bulletIndex) => (
            <li key={bulletIndex}>{bullet}</li>
          ))}
        </ul>

        {collapsible && (
          <button
            type="button"
            className={styles.showMore}
            onClick={() => toggleBlock(key)}
          >
            {expanded
              ? "Show less"
              : `Show ${block.bullets.length - VISIBLE_BULLETS} more`}
          </button>
        )}
      </section>
    );
  };

  const renderMentions = () => {
    const rows = mentionsExpanded
      ? mentions.rows
      : mentions.rows.slice(0, VISIBLE_MENTIONS);

    return (
      <section className={styles.mentions}>
        <div className={styles.mentionsHeader}>
          <div>
            <h3>Mentioned in other labels</h3>
            <p>
              Medicines whose interaction text mentions{" "}
              {medicine.name}. This is a text match, not a verified
              interaction check.
            </p>
          </div>

          {mentions.status === "ready" && (
            <span className={styles.count}>{mentions.rows.length}</span>
          )}
        </div>

        {mentions.status === "loading" && (
          <p className={styles.muted}>Checking other labels...</p>
        )}

        {mentions.status === "error" && (
          <p className={styles.muted}>
            Couldn't check other labels right now.
          </p>
        )}

        {mentions.status === "ready" && mentions.rows.length === 0 && (
          <p className={styles.muted}>
            No other labels in RxCT mention {medicine.name}.
          </p>
        )}

        {rows.length > 0 && (
          <ul className={styles.mentionList}>
            {rows.map((row) => (
              <li key={row.medicine_id}>
                <button
                  type="button"
                  className={styles.mentionItem}
                  onClick={() => onOpenMedicine(row.medicine_name)}
                >
                  <span>
                    <strong>{row.medicine_name}</strong>
                    <small>
                      {formatClassName(row.drug_class) || "Medicine"}
                    </small>
                  </span>

                  <span>→</span>
                </button>

                <p className={styles.snippet}>{row.snippet}</p>
              </li>
            ))}
          </ul>
        )}

        {mentions.rows.length > VISIBLE_MENTIONS && (
          <button
            type="button"
            className={styles.showMore}
            onClick={() => setMentionsExpanded((open) => !open)}
          >
            {mentionsExpanded
              ? "Show fewer"
              : `Show all ${mentions.rows.length}`}
          </button>
        )}
      </section>
    );
  };

  /* ========================================
     RENDER
  ======================================== */

  return (
    <div className={styles.study}>
      {(notice || matchedNote) && (
        <div className={styles.notices}>
          {notice && (
            <div className={styles.dosageNotice} role="note">
              <strong>Dosing not included</strong>
              {notice}
            </div>
          )}

          {matchedNote && (
            <div className={styles.infoNotice}>{matchedNote}</div>
          )}
        </div>
      )}

      <div className={styles.layout}>
        <div className={styles.main}>
          <div
            className={styles.tabs}
            role="tablist"
            aria-label="Medicine sections"
          >
            {availableTabs.map((key) => (
              <button
                type="button"
                role="tab"
                key={key}
                id={`study-tab-${key}`}
                aria-selected={activeTab === key}
                aria-controls="study-panel"
                className={
                  activeTab === key
                    ? `${styles.tab} ${styles.tabActive}`
                    : styles.tab
                }
                onClick={() => selectTab(key)}
              >
                {SECTION_LABELS[key]}
              </button>
            ))}
          </div>

          <article
            className={styles.panel}
            id="study-panel"
            role="tabpanel"
            aria-labelledby={`study-tab-${activeTab}`}
            ref={panelRef}
          >
            <header className={styles.panelHeader}>
              <span className={styles.eyebrow}>STUDY MATERIAL</span>
              <h2>{SECTION_LABELS[activeTab]}</h2>
            </header>

            {current.blocks.map(renderBlock)}

            {activeTab === "interactions" && renderMentions()}

            <footer className={styles.provenance}>
              <div>
                <strong>Source</strong>
                <span>{record.sources?.name || "Source not recorded"}</span>
              </div>

              {current.source && (
                <div>
                  <strong>Label</strong>
                  <span>
                    {current.source.title}, effective{" "}
                    {current.source.effective}
                  </span>

                  {current.source.url && (
                    <a
                      href={current.source.url}
                      target="_blank"
                      rel="noopener noreferrer"
                    >
                      View source label ↗
                    </a>
                  )}
                </div>
              )}

              <div>
                <strong>Review status</strong>
                <span>
                  {REVIEW_LABELS[record.review_status] ||
                    "Not specified"}
                </span>
              </div>

              <p className={styles.provenanceNote}>
                {current.truncated &&
                  "This section was shortened on import. "}
                Wording is cleaned and split into points for
                readability. Open the source label for the exact text.
              </p>
            </footer>
          </article>
        </div>

        <aside
          className={
            railOpen ? styles.rail : `${styles.rail} ${styles.railClosed}`
          }
        >
          <button
            type="button"
            className={styles.railHeader}
            aria-expanded={railOpen}
            onClick={() => setRailOpen((open) => !open)}
          >
            <span>At a Glance</span>
            <span className={styles.railToggle}>
              {railOpen ? "−" : "+"}
            </span>
          </button>

          <div className={styles.railBody}>
            {contraindicationPoints.length > 0 && (
              <section className={styles.railDanger}>
                <h3>Key contraindications</h3>

                <ul>
                  {contraindicationPoints.map((point, index) => (
                    <li key={index}>{point}</li>
                  ))}
                </ul>

                <button
                  type="button"
                  onClick={() => selectTab("contraindications")}
                >
                  All contraindications →
                </button>
              </section>
            )}

            {interactionPoints.length > 0 && (
              <section className={styles.railInfo}>
                <h3>Key interactions</h3>

                <ul>
                  {interactionPoints.map((point, index) => (
                    <li key={index}>{point}</li>
                  ))}
                </ul>

                <button
                  type="button"
                  onClick={() => selectTab("interactions")}
                >
                  All interactions →
                </button>
              </section>
            )}

            {medicine.drug_class && (
              <section className={styles.railClass}>
                <h3>Drug class</h3>

                <button
                  type="button"
                  className={styles.classChip}
                  onClick={() => onOpenClass(medicine.drug_class)}
                >
                  {formatClassName(medicine.drug_class)} →
                </button>
              </section>
            )}

            <p className={styles.railNote}>
              Study summary only. Check the source label before any
              clinical decision.
            </p>
          </div>
        </aside>
      </div>
    </div>
  );
}

export default MedicineStudy;

import styles from "./SearchResultsList.module.css";

const EXAMPLE_SEARCHES = [
  "amoxicillin",
  "antibiotic classes",
  "painkillers",
  "warfarin interactions",
];

const normalise = (value = "") => String(value).trim().toLowerCase();

function SearchResultsList({
  search,
  onOpenMedicine,
  onOpenClass,
  onOpenGroup,
  onSearch,
}) {
  const {
    kind,
    subject,
    notice,
    didYouMean,
    medicines,
    classResult,
    interactions,
  } = search;

  const hasInteractions = interactions.length > 0;
  const isEmpty = kind === "none" && !hasInteractions;

  /* ========================================
     MEDICINES
  ======================================== */

  const renderMedicines = () => (
    <section className={styles.section}>
      <div className={styles.sectionHeading}>
        <div>
          <span className={styles.eyebrow}>
            {didYouMean ? "DID YOU MEAN" : "MEDICINES"}
          </span>

          <h2>
            {didYouMean
              ? `No exact match for "${subject}"`
              : "Matching medicines"}
          </h2>
        </div>

        <span className={styles.count}>{medicines.length}</span>
      </div>

      <div className={styles.list}>
        {medicines.map((item, index) => {
          const viaOtherName =
            item.matched_term &&
            normalise(item.matched_term) !== normalise(item.name);

          return (
            <button
              type="button"
              className={styles.row}
              key={item.id}
              onClick={() => onOpenMedicine(item.name)}
            >
              <span className={styles.rowText}>
                <strong>{item.name}</strong>

                <small>
                  {formatClassName(item.drug_class) ||
                    "Class not specified"}
                </small>

                {viaOtherName && (
                  <em className={styles.tag}>
                    Matched "{item.matched_term}"
                  </em>
                )}
              </span>

              {index === 0 && !didYouMean && (
                <span className={styles.best}>Best match</span>
              )}

              <span className={styles.arrow}>→</span>
            </button>
          );
        })}
      </div>
    </section>
  );

  /* ========================================
     CLASSES AND GROUPS
  ======================================== */

  const renderClasses = () => {
    const isGroup = classResult.type === "group";

    return (
      <section className={styles.section}>
        <div className={styles.sectionHeading}>
          <div>
            <span className={styles.eyebrow}>
              {isGroup ? "THERAPEUTIC GROUP" : "DRUG CLASSES"}
            </span>

            <h2>
              {isGroup
                ? classResult.title
                : `Classes matching "${subject}"`}
            </h2>

            <p>
              {classResult.medicineCount}{" "}
              {classResult.medicineCount === 1
                ? "medicine"
                : "medicines"}{" "}
              across {classResult.classes.length}{" "}
              {classResult.classes.length === 1
                ? "drug class"
                : "drug classes"}
            </p>
          </div>

          {isGroup && (
            <button
              type="button"
              className={styles.textButton}
              onClick={() => onOpenGroup(classResult.title)}
            >
              Open group →
            </button>
          )}
        </div>

        <div className={styles.classCards}>
          {classResult.classes.map((item) => (
            <article className={styles.classCard} key={item.drugClass}>
              <button
                type="button"
                className={styles.classHeader}
                onClick={() => onOpenClass(item.drugClass)}
              >
                <strong>{formatClassName(item.drugClass)}</strong>

                <span>
                  {item.medicines.length}{" "}
                  {item.medicines.length === 1 ? "medicine" : "medicines"}{" "}
                  →
                </span>
              </button>

              <div className={styles.chips}>
                {item.medicines.map((medicine) => (
                  <button
                    type="button"
                    className={styles.chip}
                    key={medicine.id}
                    onClick={() => onOpenMedicine(medicine.name)}
                  >
                    {medicine.name}
                  </button>
                ))}
              </div>
            </article>
          ))}
        </div>
      </section>
    );

const readCache = () => {
  try {
    const value = localStorage.getItem(CACHE_KEY);
    return value ? JSON.parse(value) : null;
  } catch {
    return null;
  }
};

const writeCache = (value) => {
  try {
    localStorage.setItem(CACHE_KEY, JSON.stringify(value));
  } catch {
    // Storage may be unavailable or full.
  }
};

// RxCT uses one global search through the Topbar.
const focusGlobalSearch = () => {
  const input =
    document.querySelector("[data-global-search]") ||
    document.querySelector('header input[type="search"]');

  if (!input) return;

  input.focus();
  input.select?.();
};

const formatDate = (value) => {
  if (!value) return "";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return "";

  return date.toLocaleDateString(undefined, {
    day: "numeric",
    month: "short",
  });
};

import { useEffect, useRef, useState } from "react";
import styles from "./InstallButton.module.css";

const DISMISSED_KEY = "rxct-install-dismissed";

function InstallButton() {
  const [installPrompt, setInstallPrompt] = useState(null);

  const [installed, setInstalled] = useState(
    () =>
      window.matchMedia("(display-mode: standalone)").matches ||
      window.navigator.standalone === true
  );

  const [showOnboarding, setShowOnboarding] = useState(false);
  const [showInstructions, setShowInstructions] = useState(false);
  const [dragY, setDragY] = useState(0);
  const [isDragging, setIsDragging] = useState(false);

  const dragStartY = useRef(null);
  const dragMoved = useRef(false);

  const isIOS =
    /iPad|iPhone|iPod/.test(navigator.userAgent) ||
    (navigator.platform === "MacIntel" &&
      navigator.maxTouchPoints > 1);

  const isAndroid = /Android/i.test(navigator.userAgent);
  const isMobile = isIOS || isAndroid;

  // Installation prompt listener
useEffect(() => {
  const syncInstallPrompt = () => {
    const promptEvent = window.__rxctInstallPrompt;

    if (!promptEvent) return;

    setInstallPrompt(promptEvent);
    setShowInstructions(false);

    if (
      isAndroid &&
      !installed &&
      sessionStorage.getItem(DISMISSED_KEY) !== "true"
    ) {
      setShowOnboarding(true);
    }
  };

  const handleInstalled = () => {
    setInstalled(true);
    setShowOnboarding(false);
    setShowInstructions(false);
    setInstallPrompt(null);
    window.__rxctInstallPrompt = null;
  };

  syncInstallPrompt();

  window.addEventListener(
    "rxct-install-prompt-ready",
    syncInstallPrompt
  );

  window.addEventListener("appinstalled", handleInstalled);

  let timer;

  // iOS does not support beforeinstallprompt.
  if (
    isIOS &&
    !installed &&
    sessionStorage.getItem(DISMISSED_KEY) !== "true"
  ) {
    timer = setTimeout(() => {
      setShowOnboarding(true);
    }, 2000);
  }

  return () => {
    clearTimeout(timer);

    window.removeEventListener(
      "rxct-install-prompt-ready",
      syncInstallPrompt
    );

    window.removeEventListener("appinstalled", handleInstalled);
  };
}, [installed, isAndroid, isIOS]);

  // Prevent background scrolling
  useEffect(() => {
    if (!showOnboarding) return;

    const previousOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";

    return () => {
      document.body.style.overflow = previousOverflow;
    };
  }, [showOnboarding]);

  // Drag interaction
  function handleDragStart(event) {
    if (event.target.closest("button, a")) return;

    dragStartY.current = event.clientY;
    dragMoved.current = false;

    event.currentTarget.setPointerCapture(event.pointerId);
  }

  function handleDragMove(event) {
    if (dragStartY.current === null) return;

    const distance = Math.max(
      0,
      event.clientY - dragStartY.current
    );

    if (distance > 5) {
      dragMoved.current = true;
      setIsDragging(true);
    }

    setDragY(distance);
  }

  function handleDragEnd() {
    if (dragStartY.current === null) return;

    if (dragY > 90) {
      handleDismiss();
    }

    dragStartY.current = null;
    setIsDragging(false);
    setDragY(0);
  }

  // Installation
async function handleInstall() {
  const promptEvent =
    window.__rxctInstallPrompt || installPrompt;

  if (!promptEvent) {
    if (isIOS) {
      setShowInstructions(true);
    } else {
      setShowOnboarding(false);
    }

    return;
  }

  try {
    await promptEvent.prompt();

    const { outcome } = await promptEvent.userChoice;

    window.__rxctInstallPrompt = null;
    setInstallPrompt(null);
    setShowOnboarding(false);
    setShowInstructions(false);

    if (outcome === "accepted") {
      setInstalled(true);
    }
  } catch (error) {
    console.error("RxCT installation failed:", error);
  }
}
  function handleDismiss() {
    sessionStorage.setItem(DISMISSED_KEY, "true");
    setShowOnboarding(false);
    setDragY(0);
  }

  if (installed) return null;

  return (
    <>
      {/* Existing install button placement */}
    {!isMobile && installPrompt && (
          <button
            type="button"
            className={styles.installButton}
            onClick={() => {
              setShowInstructions(false);
              setShowOnboarding(true);
            }}
          >
            <img src="/pwa-192.png" alt="" />
            <span>Install RxCT</span>
          </button>
        )}

      {showOnboarding && (
        <div
          className={styles.overlay}
          onClick={(event) => {
            if (event.target === event.currentTarget) {
              handleDismiss();
            }
          }}
        >
          <section
            className={styles.installSheet}
            role="dialog"
            aria-modal="true"
            aria-labelledby="install-title"
            onPointerDown={handleDragStart}
            onPointerMove={handleDragMove}
            onPointerUp={handleDragEnd}
            onPointerCancel={handleDragEnd}
            onClick={(event) => {
              if (dragMoved.current) {
                event.preventDefault();
                event.stopPropagation();
                dragMoved.current = false;
              }
            }}
            style={{
              transform: `translateY(${dragY}px)`,
              transition: isDragging
                ? "none"
                : "transform 220ms ease",
            }}
          >
            <div className={styles.sheetHandle} />

            <img
              src="/pwa-512.png"
              alt="RxCT"
              className={styles.logo}
            />

            <h2 id="install-title">Make RxCT yours.</h2>

            <p>
              Keep your pharmacy workspace one tap away.
              Install RxCT on your home screen for a dedicated
              app experience and quicker access.
            </p>

            {showInstructions && (
              <div className={styles.instructions}>

                    <p>To install RxCT on your iPhone or iPad:</p>

                      <li>Tap the Share button.</li>
                      <li>Select Add to Home Screen.</li>
                      <li>Tap Add.</li>

                    <ol>

                      <li>Select Install app or Add to Home screen.</li>

const pageIcon = location.pathname === "/medicines" ? "fa-capsules" : location.pathname === "/resources" ? "fa-folder-open" : location.pathname === "/history" ? "fa-clock-rotate-left" : "fa-table-columns";

useEffect(() => {
if (!menuOpen) return;

function handleClickOutside(event) {
  if (menuRef.current && !menuRef.current.contains(event.target)) {
    setMenuOpen(false);
  }
}

document.addEventListener("mousedown", handleClickOutside);

return () => {
  document.removeEventListener("mousedown", handleClickOutside);
};

}, [menuOpen]);
useEffect(() => {
  let active = true;

  const query = text.trim();

  const timer = setTimeout(async () => {
    if (query.length < 2) {
      setSuggestions([]);
      setSuggestionsLoading(false);
      return;
    }

    try {
      setSuggestionsLoading(true);

      const data = await searchSuggestions(query);

      if (active) {
        setSuggestions(data || []);
      }
    } catch (error) {
      console.error("Autocomplete error:", error);

      if (active) setSuggestions([]);
    } finally {
      if (active) setSuggestionsLoading(false);
    }
  }, 250);

  return () => {
    active = false;
    clearTimeout(timer);
  };
}, [text]);
function handleSearchKeyDown(event) {
if (event.key === "Enter" && text.trim()) {
navigate("/medicines?q=" + encodeURIComponent(text.trim()));
setSearchExpanded(false);
searchRef.current?.blur();
}

if (event.key === "Escape") {
  setSearchExpanded(false);
  searchRef.current?.blur();
}

}

return (
<header
className={`${styles.topbar} ${
  searchExpanded ? styles.searchExpanded : ""
}`}>
<div className={styles.brand}>
<img src={logo} alt="RxCT" />
</div>

  <h2 className={styles.pageTitle}>
    <i className={`fa-solid ${pageIcon}`}></i>
    {pageTitle}
  </h2>

  <div className={styles.search}>
    <i className="fa-solid fa-magnifying-glass"></i>

    <input
      ref={searchRef}
      data-global-search
      className={styles.searchInput}
      value={text}
      type="search"
      placeholder="Search medicines,classes....."
onFocus={() => {
  setSearchExpanded(true);
  setSearchFocused(true);
}}
onBlur={() => {
  setSearchExpanded(false);
  setSearchFocused(false);
}}
      onChange={(e) => setText(e.target.value)}
         onKeyDown={handleSearchKeyDown}
    />

    {searchFocused && text.trim().length >= 2 && (
      <div className={styles.suggestions}>
        {suggestionsLoading ? (
          <div className={styles.suggestionMessage}>
            Searching...
          </div>
        ) : suggestions.length > 0 ? (
          suggestions.slice(0, 4).map((item) => (
            <button
              key={`${item.suggestion_type}-${item.target_value}`}
              className={styles.suggestionItem}
              type="button"
              onMouseDown={(event) => event.preventDefault()}
              onClick={() => {
                setText(item.label);
                setSearchFocused(false);
                setSearchExpanded(false);
                searchRef.current?.blur();

                navigate(
                  "/medicines?q=" +
                    encodeURIComponent(item.label)
                );
              }}
            >
              <span className={styles.suggestionIcon}>
                <i
                  className={
                    item.suggestion_type === "medicine"
                      ? "fa-solid fa-capsules"
                      : "fa-solid fa-layer-group"
                  }
                ></i>
              </span>

              <span className={styles.suggestionDetails}>
                <strong>{item.label}</strong>
                <small>{item.detail}</small>
              </span>

              <i className="fa-solid fa-arrow-up-right-from-square"></i>
            </button>
          ))
        ) : (
          <div className={styles.suggestionMessage}>
            No suggestions found.
          </div>
        )}
      </div>
    )}
  </div>

  <div className={styles.actions}>
  {!panelOpen && (
    <>
      <InstallButton />

      <button className={styles.iconButton} aria-label="Notifications">
        <i className="fa-solid fa-bell"></i>
      </button>
    </>
  )}
</div>

  {!panelOpen && (
    <div className={styles.profileMenu} ref={menuRef}>
      <button
        className={styles.avatarButton}
        aria-label="Open profile menu"
        aria-expanded={menuOpen}
        onClick={() => setMenuOpen(!menuOpen)}
      >
        <img src={avatar} alt="" className={styles.avatarImage} />
      </button>

      {menuOpen && (
        <div className={styles.dropdown}>
          <Link
            to="/profile"
            onClick={(event) => {
              event.preventDefault();
              onOpenPanel("profile");
              setMenuOpen(false);
            }}
          >
            Profile
          </Link>

          <hr className={styles.divider} />

          <Link
            to="/settings"
            onClick={(event) => {
              event.preventDefault();
              onOpenPanel("settings");
              setMenuOpen(false);
            }}
          >
            Settings
          </Link>
        </div>
      )}
    </div>
  )}
</header>

);
}

import { useCallback, useEffect, useRef, useState } from "react";
import { useNavigate, useSearchParams } from "react-router-dom";

import MedicinesLanding from "../components/medicines/MedicinesLanding";
import SearchResultsList from "../components/medicines/SearchResultsList";
import MedicineProfile from "../components/study/MedicineProfile";
import ClassProfile from "../components/study/ClassProfile";
import GroupProfile from "../components/study/GroupProfile";
import Jotter from "../components/jotter/Jotter";

import { resolveQuery } from "../services/profiles";
import { recordVisit, saveResume } from "../services/activity";
import { useScrollMemory } from "../hooks/useScrollMemory";
import { formatClassName } from "../utils/labelText";

import styles from "./Medicines.module.css";

/* ========================================
   MEDICINES
   One page, four views, decided by the URL:

     /medicines                       landing
     /medicines?q=beta blockers       search results
     /medicines?kind=medicine&q=...   a profile
       (&section=...&sub=...)         opened at a place

   Every step is a normal history entry, so
   Back always returns to the previous
   search, class or medicine.
======================================== */

function MedicinesView() {
  const [params, setParams] = useSearchParams();
  const navigate = useNavigate();

  const query = params.get("q") || "";
  const kind = params.get("kind") || "";
  const section = params.get("section") || "";
  const sub = params.get("sub") || "";

 const [view, setView] = useState(null);

  const pageRef = useRef(null);

  /* ========================================
     RESOLVE WHAT THE URL POINTS AT
  ======================================== */

 useEffect(() => {
  if (!query.trim()) return;

  let active = true;


  resolveQuery(query, kind)
    .then((result) => {
      if (active) {
        setView({ status: "ready", ...result });
      }
    })
    .catch((error) => {
      if (active) {
        setView({
          status: "error",
          message: error?.message || "Unable to load this page.",
        });
      }
    })

  return () => {
    active = false;
  };
}, [query, kind]);
const isLanding = !query.trim();

const isLoading =
  !isLanding &&
  (!view || view.status === "loading");

  /* ========================================
     ACTIVITY
  ======================================== */

useEffect(() => {
  if (view?.status === "ready" && view.type !== "search") {
    recordVisit({ type: view.type, name: view.name });
  }
}, [view]);
  const handleSectionChange = useCallback(
    (sectionKey, label) => {
      if (view.status !== "ready" || view.type === "search") return;

      saveResume({
        destination: new URLSearchParams({
          kind: view.type,
          q: view.name,
          section: sectionKey,
        }).toString(),
        title:
          view.type === "class" ? formatClassName(view.name) : view.name,
        section: label,
      });
    },
    [view],
  );

  /* ========================================
     NAVIGATION
  ======================================== */

  const open = useCallback(
    (type, name) => setParams({ kind: type, q: name }),
    [setParams],
  );

  const openMedicine = useCallback(
    (name) => open("medicine", name),
    [open],
  );

  const openClass = useCallback((name) => open("class", name), [open]);
  const openGroup = useCallback((name) => open("group", name), [open]);

  const runSearch = useCallback(
    (text) => setParams({ q: text }),
    [setParams],
  );

  // Back goes to wherever the student came from inside RxCT.
  // Opened directly (a bookmark, a reload), it goes to the landing page.
  const goBack = useCallback(() => {
    if (window.history.state?.idx > 0) navigate(-1);
    else navigate("/medicines");
  }, [navigate]);

  /* ========================================
     SCROLL MEMORY FOR LANDING AND SEARCH
     (profiles remember their own)
  ======================================== */
const isSearch = view?.status === "ready" && view.type === "search";

  useScrollMemory(
    pageRef,
    isSearch ? `search:${query.toLowerCase()}` : "landing",
    isSearch || isLanding,
  );

  /* ========================================
     RENDER
  ======================================== */

  if (isLanding) {
    return (
      <div ref={pageRef}>
        <MedicinesLanding
          onOpenClass={openClass}
          onOpenGroup={openGroup}
          onOpenMedicine={openMedicine}
          onResume={(destination) => setParams(destination)}
        />
      </div>
    );
  }

if (isLoading) {
    return (
      <main className={styles.workspace}>
        <div className={styles.loading} role="status">
          <span className={styles.loader}></span>
          <p>Opening...</p>
        </div>
      </main>
    );
  }

  if (view?.status === "error") {
    return (
      <main className={styles.workspace}>
        <div className={styles.error}>
          <h2>Something went wrong</h2>
          <p>{view.message}</p>

          <button type="button" onClick={() => navigate("/medicines")}>
            Return to Medicines
          </button>
        </div>
      </main>
    );
  }

  if (isSearch) {
    return (
      <main className={styles.workspace} ref={pageRef}>
        <section className={styles.searchResults}>
          <button
            type="button"
            className={styles.backButton}
            onClick={goBack}
          >
            ← Back
          </button>

          <h1>{query}</h1>

          <SearchResultsList
            search={view.found}
            onOpenMedicine={openMedicine}
            onOpenClass={openClass}
            onOpenGroup={openGroup}
            onSearch={runSearch}
          />
        </section>
      </main>
    );
  }

  const initialSection = section || view.openSection || "";
  const profileKey = `${view.type}:${view.name}`;

  return (
    <main className={styles.workspace}>
      {view.type === "medicine" && (
        <MedicineProfile
          key={profileKey}
          medicine={view.data}
          initialSection={initialSection}
          initialSub={sub}
          notice={view.notice}
          matchedNote={view.matchedNote}
          onBack={goBack}
          onOpenClass={openClass}
          onOpenGroup={openGroup}
          onOpenMedicine={openMedicine}
          onSectionChange={handleSectionChange}
        />
      )}

      {view.type === "class" && (
        <ClassProfile
          key={profileKey}
          drugClass={view.data}
          initialSection={initialSection}
          initialSub={sub}
          onBack={goBack}
          onOpenMedicine={openMedicine}
          onOpenGroup={openGroup}
          onSectionChange={handleSectionChange}
        />
      )}

      {view.type === "group" && (
        <GroupProfile
          key={profileKey}
          group={view.data}
          initialSection={initialSection}
          initialSub={sub}
          onBack={goBack}
          onOpenClass={openClass}
          onSectionChange={handleSectionChange}
        />
      )}
    </main>
  );
}

function History() {
  const navigate = useNavigate();
  const [items, setItems] = useState(() => {
    try { return JSON.parse(localStorage.getItem("rxct-medicine-history") || "[]"); }
    catch { return []; }
  });

  useEffect(() => {
    function refresh() {
      try { setItems(JSON.parse(localStorage.getItem("rxct-medicine-history") || "[]")); }
      catch { setItems([]); }
    }
    window.addEventListener("storage", refresh);
    return () => window.removeEventListener("storage", refresh);
  }, []);

  return (
    <div className={styles.page}>
      <span className={styles.eyebrow}>YOUR ACTIVITY</span>
      <h1>History</h1>
      <p className={styles.intro}>Pick up from the medicines you have recently opened.</p>
      {items.length > 0 ? (
        <div className={styles.list}>
          {items.map((item) => (

              <span className={styles.details}><strong>{item.name}</strong><small>{item.drugClass || "Medicine reference"}</small></span>

import { useCallback, useState } from "react";
import MedicinesWorkspaceContext from "./MedicinesWorkspace";
export function MedicinesWorkspaceProvider({ children }) {
  const [history, setHistory] = useState([
    { type: "library" },
  ]);

  const current = history[history.length - 1];
  const navigateTo = useCallback((destination) => {
    setHistory((previous) => [...previous, destination]);
  }, []);

  const goBack = useCallback(() => {
    setHistory((previous) =>
      previous.length > 1
        ? previous.slice(0, -1)
        : previous,
    );
  }, []);

  const resetWorkspace = useCallback(() => {
    setHistory([{ type: "library" }]);
  }, []);

  const replaceWorkspace = useCallback((destination) => {
    setHistory([destination]);
  }, []);
  return (
    <MedicinesWorkspaceContext.Provider
  value={{
  history,
  current,
  navigateTo,
  goBack,
  resetWorkspace,
  replaceWorkspace,
}}
    >
      {children}
    </MedicinesWorkspaceContext.Provider>
  );
}
